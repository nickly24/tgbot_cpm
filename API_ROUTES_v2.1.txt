📝 1. РЕДАКТИРОВАНИЕ ИМЕНИ ПОЛЬЗОВАТЕЛЯ (УЛУЧШЕНО)
═══════════════════════════════════════════════════════════════════

Endpoint: PUT /api/chats/<chat_id>/name

Описание: Обновить имя пользователя в чате

Параметры URL:
  - chat_id (integer) - ID чата пользователя

Тело запроса (JSON):
  {
    "name": "Новое Имя"  // обязательно, не может быть пустым
  }

Пример запроса:
  PUT http://109.73.202.73:80/api/chats/123456789/name
  Content-Type: application/json
  
  {
    "name": "Иван Петров"
  }

Успешный ответ (200):
  {
    "success": true,
    "message": "Имя пользователя обновлено успешно",
    "new_name": "Иван Петров"
  }

Возможные ошибки:
  
  404 Not Found - Пользователь не найден
  {
    "success": false,
    "error": "Пользователь не найден"
  }
  
  400 Bad Request - Поле name отсутствует или пустое
  {
    "success": false,
    "error": "Поле 'name' обязательно"
  }
  
  400 Bad Request - Имя пустое (пробелы)
  {
    "success": false,
    "error": "Имя не может быть пустым"
  }

🗑️ 2. УДАЛЕНИЕ ЧАТА (НОВОЕ)
═══════════════════════════════════════════════════════════════════

Endpoint: DELETE /api/chats/<chat_id>

Описание: Полное удаление чата со всеми сообщениями

Параметры URL:
  - chat_id (integer) - ID чата для удаления

Тело запроса: не требуется

Пример запроса:
  DELETE http://109.73.202.73:80/api/chats/123456789

Успешный ответ (200):
  {
    "success": true,
    "message": "Чат успешно удален",
    "deleted": {
      "user": true,                // был ли удален пользователь
      "messages_count": 42         // количество удаленных сообщений
    }
  }

Возможные ошибки:
  
  404 Not Found - Чат не найден
  {
    "success": false,
    "error": "Чат не найден"
  }
  
  500 Internal Server Error - Ошибка при удалении
  {
    "success": false,
    "error": "Ошибка при удалении чата"
  }

⚠️ ВАЖНО:
  - Операция необратима
  - Удаляет пользователя из БД
  - Удаляет ВСЕ сообщения этого чата
  - Рекомендуется добавить подтверждение в UI

📊 ИЗМЕНЕНИЯ В ДАННЫХ
═══════════════════════════════════════════════════════════════════

Поле "name" в пользователе:
  - РАНЬШЕ: Бралось из Telegram профиля (first_name + last_name)
  - СЕЙЧАС: Вводится пользователем при первом контакте с ботом
  - Может быть изменено через API

Новый флоу регистрации пользователя в боте:
  1. /start
  2. Пользователь вводит своё имя
  3. Пользователь выбирает статус (студент/абитуриент/родитель)
  4. Готово - может писать сообщения

JavaScript/Fetch:

// Редактирование имени
async function updateUserName(chatId, newName) {
  const response = await fetch(`http://109.73.202.73:80/api/chats/${chatId}/name`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: newName })
  });
  return await response.json();
}

// Удаление чата
async function deleteChat(chatId) {
  const response = await fetch(`http://109.73.202.73:80/api/chats/${chatId}`, {
    method: 'DELETE'
  });
  return await response.json();
}

// Использование
try {
  // Изменить имя
  const result = await updateUserName(123456789, "Новое Имя");
  if (result.success) {
    console.log('Имя изменено:', result.new_name);
  }
  
  // Удалить чат (с подтверждением)
  if (confirm('Удалить чат? Это действие нельзя отменить!')) {
    const result = await deleteChat(123456789);
    if (result.success) {
      console.log(`Удалено ${result.deleted.messages_count} сообщений`);
    }
  }
} catch (error) {
  console.error('Ошибка:', error);
}


Axios:

// Редактирование имени

  try {

  try {
    const { data } = await axios.delete(
      `http://109.73.202.73:80/api/chats/${chatId}`
    );
    return data;
  } catch (error) {
    console.error(error.response?.data || error.message);
    throw error;
  }
};

// Форма редактирования имени
function EditNameForm({ chatId, currentName, onUpdate }) {
  const [name, setName] = useState(currentName);
  const [loading, setLoading] = useState(false);
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!name.trim()) {
      alert('Имя не может быть пустым');
      return;
    }
    
    setLoading(true);
    try {
      const result = await updateUserName(chatId, name.trim());
      if (result.success) {
        onUpdate(result.new_name);
        alert('Имя обновлено!');
      }
    } catch (error) {
      alert('Ошибка обновления имени');
    } finally {
      setLoading(false);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input 
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="Введите имя"
      />
      <button type="submit" disabled={loading}>
        {loading ? 'Сохранение...' : 'Сохранить'}
      </button>
    </form>
  );
}

// Кнопка удаления чата
function DeleteChatButton({ chatId, userName, messagesCount, onDelete }) {
  const [showModal, setShowModal] = useState(false);
  const [deleting, setDeleting] = useState(false);
  
  const handleDelete = async () => {
    setDeleting(true);
    try {
      const result = await deleteChat(chatId);
      if (result.success) {
        onDelete(chatId);
        setShowModal(false);
        alert(`Чат удален (${result.deleted.messages_count} сообщений)`);
      }
    } catch (error) {
      alert('Ошибка удаления');
    } finally {
      setDeleting(false);
    }
  };
  
  return (
    <>
      <button 
        onClick={() => setShowModal(true)}
        className="btn-danger"
      >
        🗑️ Удалить чат
      </button>
      
      {showModal && (
        <Modal>
          <h3>⚠️ Удалить чат?</h3>
          <p>Пользователь: <strong>{userName}</strong></p>
          <p>Сообщений: <strong>{messagesCount}</strong></p>
          <p className="warning">
            Это действие нельзя отменить!
          </p>
          <button 
            onClick={handleDelete}
            disabled={deleting}
            className="btn-danger"
          >
            {deleting ? 'Удаление...' : 'Да, удалить'}
          </button>
          <button onClick={() => setShowModal(false)}>
            Отмена
          </button>
        </Modal>
      )}
    </>
  );
}
