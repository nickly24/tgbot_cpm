1. ОБЗОР ПРОЕКТА
================================================================================

ОПИСАНИЕ:
Веб-админка для управления Telegram ботом. Администраторы могут:
- Просматривать все чаты с клиентами
- Отвечать на сообщения клиентов
- Видеть статус клиента (студент, абитуриент, родитель)
- Отслеживать непрочитанные сообщения
- Менять статус клиента
- Делать массовую рассылку по группам

ТЕХНОЛОГИИ:
- Frontend: React.js
- Backend: Python Flask REST API
- База данных: MongoDB
- Telegram Bot: python-telegram-bot

API BASE URL: http://0.0.0.0:80/api

// api/client.js
const API_BASE_URL = 'http://0.0.0.0:80/api';

class ApiClient {
  async request(endpoint, options = {}) {
    const url = `${API_BASE_URL}${endpoint}`;
    const config = {
      headers: {
        'Content-Type': 'application/json',
        ...options.headers,
      },
      ...options,
    };

    const response = await fetch(url, config);
    const data = await response.json();

    if (!data.success) {
      throw new Error(data.error || 'API Error');
    }

    return data;
  }

  // Чаты
  getChats() {
    return this.request('/chats');
  }

  getChatHistory(chatId) {
    return this.request(`/chats/${chatId}`);
  }

  sendMessage(chatId, message, adminName = 'Администратор') {
    return this.request(`/chats/${chatId}/send`, {
      method: 'POST',
      body: JSON.stringify({ message, admin_name: adminName }),
    });
  }

  markAsRead(chatId) {
    return this.request(`/chats/${chatId}/mark-read`, {
      method: 'POST',
    });
  }

  // Статусы
  getStatuses() {
    return this.request('/statuses');
  }

  updateStatus(chatId, status) {
    return this.request(`/chats/${chatId}/status`, {
      method: 'PUT',
      body: JSON.stringify({ status }),
    });
  }

  // Рассылка
  sendBroadcast(message, statuses = null, adminName = 'Администратор') {
    const body = { message, admin_name: adminName };
    if (statuses) body.statuses = statuses;
    
    return this.request('/broadcast', {
      method: 'POST',
      body: JSON.stringify(body),
    });
  }

  // Статистика
  getStats() {
    return this.request('/stats');
  }
}

export const apiClient = new ApiClient();

// hooks/useChats.js
import { useState, useEffect } from 'react';
import { apiClient } from '../api/client';

export const useChats = () => {
  const [chats, setChats] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const fetchChats = async () => {
    try {
      setLoading(true);
      const response = await apiClient.getChats();
      setChats(response.data);
      setError(null);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchChats();
    
    // Автообновление каждые 5 секунд
    const interval = setInterval(fetchChats, 5000);
    return () => clearInterval(interval);
  }, []);

  return { chats, loading, error, refetch: fetchChats };
};

// components/ChatWindow.jsx
import React, { useState, useEffect, useRef } from 'react';
import { apiClient } from '../api/client';

const ChatWindow = ({ chatId }) => {
  const [messages, setMessages] = useState([]);
  const [newMessage, setNewMessage] = useState('');
  const [userInfo, setUserInfo] = useState(null);
  const [status, setStatus] = useState(null);
  const messagesEndRef = useRef(null);

  const fetchHistory = async () => {
    if (!chatId) return;
    
    const response = await apiClient.getChatHistory(chatId);
    setMessages(response.data.messages);
    setUserInfo(response.data.user_info);
    setStatus(response.data.status);
  };

  useEffect(() => {
    if (!chatId) return;
    
    // Загружаем историю
    fetchHistory();
    
    // Помечаем как прочитанное
    apiClient.markAsRead(chatId);
    
    // Автообновление каждые 3 секунды
    const interval = setInterval(fetchHistory, 3000);
    return () => clearInterval(interval);
  }, [chatId]);

  useEffect(() => {
    // Автоскролл к последнему сообщению
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = async (e) => {
    e.preventDefault();
    if (!newMessage.trim()) return;

    await apiClient.sendMessage(chatId, newMessage);
    setNewMessage('');
    fetchHistory();
  };

  if (!chatId) return <div>Выберите чат</div>;

  return (
    <div className="chat-window">
      <div className="chat-header">
        <h3>{userInfo?.name}</h3>
        <span className="username">@{userInfo?.username}</span>
        <span className={`status-badge status-${status}`}>{status}</span>
      </div>

      <div className="messages">
        {messages.map(msg => (
          <div 
            key={msg._id}
            className={`message ${msg.from_user ? 'user' : 'admin'} ${!msg.is_read && msg.from_user ? 'unread' : ''}`}
          >
            <div className="message-header">
              <span className="sender">
                {msg.from_user ? userInfo?.name : msg.admin_name}
              </span>
              <span className="time">
                {new Date(msg.timestamp).toLocaleTimeString()}
              </span>
            </div>
            <div className="message-text">{msg.text}</div>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      <form onSubmit={handleSend} className="message-form">
        <input
          type="text"
          value={newMessage}
          onChange={(e) => setNewMessage(e.target.value)}
          placeholder="Введите сообщение..."
        />
        <button type="submit" disabled={!newMessage.trim()}>
          Отправить
        </button>
      </form>
    </div>
  );
};

export default ChatWindow;

    setLoading(true);
    try {
      const response = await apiClient.sendBroadcast(
        message,
        selectedStatuses.length > 0 ? selectedStatuses : null
      );
      setResult(response);
      setMessage('');
    } catch (error) {
      alert('Ошибка: ' + error.message);
    } finally {
      setLoading(false);
    }
  };

  const toggleStatus = (status) => {
    if (selectedStatuses.includes(status)) {
      setSelectedStatuses(selectedStatuses.filter(s => s !== status));
    } else {
      setSelectedStatuses([...selectedStatuses, status]);
    }
  };

  return (
    <div className="broadcast-form">
      <h2>Массовая рассылка</h2>

      <div className="target-groups">
        <h3>Кому отправить:</h3>
        {STATUSES.map(status => (
          <label key={status.value}>
            <input
              type="checkbox"
              checked={selectedStatuses.includes(status.value)}
              onChange={() => toggleStatus(status.value)}
            />
            {status.label}
          </label>
        ))}
        {selectedStatuses.length === 0 && (
          <p><em>Будет отправлено всем пользователям</em></p>
        )}
      </div>

      <textarea
        value={message}
        onChange={(e) => setMessage(e.target.value)}
        placeholder="Текст сообщения..."
        rows={5}
      />

      <button onClick={handleSend} disabled={loading || !message.trim()}>
        {loading ? 'Отправка...' : 'Отправить рассылку'}
      </button>

      {result && (
        <div className="result">
          <h3>Результат рассылки:</h3>
          <p>Отправлено: {result.stats.success} из {result.stats.total_target}</p>
          <p>Ошибок: {result.stats.failed}</p>
        </div>
      )}
    </div>
  );
};

export default BroadcastForm;

8. FAQ (ЧАСТО ЗАДАВАЕМЫЕ ВОПРОСЫ)
================================================================================

Q: Как часто обновлять список чатов?
A: Каждые 5 секунд (UPDATE_INTERVALS.CHAT_LIST)

Q: Нужна ли авторизация в API?
A: В текущей версии нет. Для production добавьте JWT токены.

Q: Как определить, что пришло новое сообщение?
A: Сравнивайте unread_count при каждом обновлении списка чатов.

Q: Можно ли отправлять Markdown в сообщениях?
A: В текущей версии нет, только plain text.

Q: Как обрабатывать offline режим?
A: Показывайте баннер "Нет связи с сервером" и прекращайте автообновление.

Q: Поддерживает ли API пагинацию?
A: В текущей версии нет. Все сообщения возвращаются сразу.

Q: Как узнать, печатает ли клиент сейчас?
A: Telegram API не предоставляет эту информацию через бота.

Q: Можно ли удалять сообщения?
A: В текущей версии нет такой возможности.
